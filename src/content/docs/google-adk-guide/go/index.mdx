## Install, minimal example

```go
package main

import (
	"context"
	"fmt"
	"log"
	"os"

	"google.golang.org/genai"

	"google.golang.org/adk/agent"
	"google.golang.org/adk/agent/llmagent"
	"google.golang.org/adk/model/gemini"
	"google.golang.org/adk/runner"
	"google.golang.org/adk/session"
)

func main() {
	ctx := context.Background()

	model, err := gemini.NewModel(ctx, "%%model=gemini%%", &genai.ClientConfig{
		APIKey: os.Getenv("GOOGLE_API_KEY"),
	})
	if err != nil {
		log.Fatal(err)
	}

	assistant, err := llmagent.New(llmagent.Config{
		Name:        "assistant",
		Model:       model,
		Instruction: "You are a helpful assistant.",
	})
	if err != nil {
		log.Fatal(err)
	}

	r, err := runner.New(runner.Config{
		AppName:           "demo",
		Agent:             assistant,
		SessionService:    session.InMemoryService(),
		AutoCreateSession: true,
	})
	if err != nil {
		log.Fatal(err)
	}

	msg := genai.NewContentFromText("What is 15 + 27?", genai.RoleUser)
	for event, err := range r.Run(ctx, "u1", "s1", msg, agent.RunConfig{}) {
		if err != nil {
			log.Fatal(err)
		}
		if event.IsFinalResponse() && event.Content != nil {
			fmt.Println(event.Content.Parts[0].Text)
		}
	}
}
```
